# 课程学习笔记 - 基环树

## A. Island

每个连通块的边数与点数相同，因此都是一棵**基环树**。渡船只能在不同连通块之间使用，所以答案是每个连通块最长简单路径之和；先拓扑删叶计算各环点挂树的最长链与直径，再将环复制两遍，用单调队列求经过环的最长路径。

代码：

```cpp
#include<bits/stdc++.h>
using namespace std;
const int M = 1000010;
int n, q[M * 2], cl[M], dg[M], ec, ew[M], hd[M], nx[M * 2], to[M * 2], wt[M * 2];
long long ff[M], hh[M], ps[M * 2];
bool cy[M], vs[M];
void add(int u, int v, int w) {
	to[ec] = v;
	wt[ec] = w;
	nx[ec] = hd[u];
	hd[u] = ec++;
}
int main() {
	ios::sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	cin >> n;
	fill(hd, hd + n + 1, -1);
	for (int i = 1, u, w; i <= n; i++) {
		cin >> u >> w;
		add(i, u, w);
		add(u, i, w);
		dg[i]++;
		dg[u]++;
	}
	int l = 0, r = 0;
	for (int i = 1; i <= n; i++) {
		if (dg[i] == 1) q[r++] = i;
	}
	while (l < r) {
		int u = q[l++], e = hd[u];
		while (e != -1 && !dg[to[e]]) e = nx[e];
		int v = to[e];
		long long z = ff[u] + wt[e];
		hh[v] = max(hh[v], max(hh[u], ff[v] + z));
		ff[v] = max(ff[v], z);
		dg[u] = 0;
		dg[v]--;
		if (dg[v] == 1) q[r++] = v;
	}
	for (int i = 1; i <= n; i++) {
		if (dg[i]) cy[i] = 1;
	}
	long long ans = 0;
	for (int s = 1; s <= n; s++) if (cy[s] && !vs[s]) {
			int k = 0, u = s, pe = -1;
			while (1) {
				vs[u] = 1;
				cl[k] = u;
				int e = hd[u];
				while (e != -1 && (!cy[to[e]] || (e >> 1) == pe)) e = nx[e];
				ew[k] = wt[e];
				pe = e >> 1;
				u = to[e];
				k++;
				if (u == s) break;
			}
			long long p = 0, res = 0;
			for (int i = 0; i < k; i++) res = max(res, hh[cl[i]]);
			l = r = 0;
			for (int j = 0; j < 2 * k; j++) {
				while (l < r && q[l] < j - k + 1) l++;
				int u = cl[j % k];
				if (l < r) res = max(res, ff[u] + p + ps[l]);
				long long z = ff[u] - p;
				while (l < r && ps[r - 1] <= z) r--;
				q[r] = j;
				ps[r++] = z;
				p += ew[j % k];
			}
			ans += res;
		}
	cout << ans << '\n';
	return 0;
}
```

## B. 骑士

把“互相不能同时选择”看成无向边，图同样是基环树森林。删叶时维护树形独立集 DP：`f0` 表示不选当前点，`f1` 表示选择当前点；剩余环分别枚举第一个骑士不选、选择两种情况进行链式 DP。

代码：

```cpp
#include<bits/stdc++.h>
using namespace std;
const int M = 1000010;
const long long INF = 4000000000000000000LL;
int n, q[M], aa[M], cl[M], dg[M], ec, hd[M], nx[M * 2], to[M * 2];
long long f0[M], f1[M];
bool cy[M], vs[M];
void add(int u, int v) {
	to[ec] = v;
	nx[ec] = hd[u];
	hd[u] = ec++;
}
int main() {
	ios::sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	cin >> n;
	fill(hd, hd + n + 1, -1);
	for (int i = 1, v; i <= n; i++) {
		cin >> aa[i] >> v;
		add(i, v);
		add(v, i);
		dg[i]++;
		dg[v]++;
		f1[i] = aa[i];
	}
	int l = 0, r = 0;
	for (int i = 1; i <= n; i++) if (dg[i] == 1) q[r++] = i;
	while (l < r) {
		int u = q[l++], e = hd[u];
		while (e != -1 && !dg[to[e]]) e = nx[e];
		int v = to[e];
		f0[v] += max(f0[u], f1[u]);
		f1[v] += f0[u];
		dg[u] = 0;
		dg[v]--;
		if (dg[v] == 1) q[r++] = v;
	}
	for (int i = 1; i <= n; i++) {
		if (dg[i]) cy[i] = 1;
	}
	long long ans = 0;
	for (int s = 1; s <= n; s++) if (cy[s] && !vs[s]) {
			int k = 0, u = s, pe = -1;
			while (1) {
				vs[u] = 1;
				cl[k++] = u;
				int e = hd[u];
				while (e != -1 && (!cy[to[e]] || (e >> 1) == pe)) e = nx[e];
				pe = e >> 1;
				u = to[e];
				if (u == s) break;
			}
			long long x = f0[cl[0]], y = -INF, z;
			for (int i = 1; i < k; i++) {
				long long a = max(x, y) + f0[cl[i]], b = x + f1[cl[i]];
				x = a;
				y = b;
			}
			z = max(x, y);
			x = -INF;
			y = f1[cl[0]];
			for (int i = 1; i < k; i++) {
				long long a = max(x, y) + f0[cl[i]], b = x + f1[cl[i]];
				x = a;
				y = b;
			}
			ans += max(z, x);
		}
	cout << ans << '\n';
	return 0;
}
```
