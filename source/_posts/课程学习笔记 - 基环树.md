# 课程学习笔记 - 基环树

## A. Island

每个连通块的边数与点数相同，因此都是一棵**基环树**。渡船只能在不同连通块之间使用，所以答案是每个连通块最长简单路径之和；先拓扑删叶计算各环点挂树的最长链与直径，再将环复制两遍，用单调队列求经过环的最长路径。

代码：

```cpp
#include<bits/stdc++.h>
using namespace std;
const int M = 1000010;
int n, q[M * 2], cl[M], dg[M], ec, ew[M], hd[M], nx[M * 2], to[M * 2], wt[M * 2];
long long ff[M], hh[M], ps[M * 2];
bool cy[M], vs[M];
void add(int u, int v, int w) {
	to[ec] = v;
	wt[ec] = w;
	nx[ec] = hd[u];
	hd[u] = ec++;
}
int main() {
	ios::sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	cin >> n;
	fill(hd, hd + n + 1, -1);
	for (int i = 1, u, w; i <= n; i++) {
		cin >> u >> w;
		add(i, u, w);
		add(u, i, w);
		dg[i]++;
		dg[u]++;
	}
	int l = 0, r = 0;
	for (int i = 1; i <= n; i++) {
		if (dg[i] == 1) q[r++] = i;
	}
	while (l < r) {
		int u = q[l++], e = hd[u];
		while (e != -1 && !dg[to[e]]) e = nx[e];
		int v = to[e];
		long long z = ff[u] + wt[e];
		hh[v] = max(hh[v], max(hh[u], ff[v] + z));
		ff[v] = max(ff[v], z);
		dg[u] = 0;
		dg[v]--;
		if (dg[v] == 1) q[r++] = v;
	}
	for (int i = 1; i <= n; i++) {
		if (dg[i]) cy[i] = 1;
	}
	long long ans = 0;
	for (int s = 1; s <= n; s++) if (cy[s] && !vs[s]) {
			int k = 0, u = s, pe = -1;
			while (1) {
				vs[u] = 1;
				cl[k] = u;
				int e = hd[u];
				while (e != -1 && (!cy[to[e]] || (e >> 1) == pe)) e = nx[e];
				ew[k] = wt[e];
				pe = e >> 1;
				u = to[e];
				k++;
				if (u == s) break;
			}
			long long p = 0, res = 0;
			for (int i = 0; i < k; i++) res = max(res, hh[cl[i]]);
			l = r = 0;
			for (int j = 0; j < 2 * k; j++) {
				while (l < r && q[l] < j - k + 1) l++;
				int u = cl[j % k];
				if (l < r) res = max(res, ff[u] + p + ps[l]);
				long long z = ff[u] - p;
				while (l < r && ps[r - 1] <= z) r--;
				q[r] = j;
				ps[r++] = z;
				p += ew[j % k];
			}
			ans += res;
		}
	cout << ans << '\n';
	return 0;
}
```

## B. 骑士

把“互相不能同时选择”看成无向边，图同样是基环树森林。删叶时维护树形独立集 DP：`f0` 表示不选当前点，`f1` 表示选择当前点；剩余环分别枚举第一个骑士不选、选择两种情况进行链式 DP。

代码：

```cpp
#include<bits/stdc++.h>
using namespace std;
const int M = 1000010;
const long long INF = 4000000000000000000LL;
int n, q[M], aa[M], cl[M], dg[M], ec, hd[M], nx[M * 2], to[M * 2];
long long f0[M], f1[M];
bool cy[M], vs[M];
void add(int u, int v) {
	to[ec] = v;
	nx[ec] = hd[u];
	hd[u] = ec++;
}
int main() {
	ios::sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	cin >> n;
	fill(hd, hd + n + 1, -1);
	for (int i = 1, v; i <= n; i++) {
		cin >> aa[i] >> v;
		add(i, v);
		add(v, i);
		dg[i]++;
		dg[v]++;
		f1[i] = aa[i];
	}
	int l = 0, r = 0;
	for (int i = 1; i <= n; i++) if (dg[i] == 1) q[r++] = i;
	while (l < r) {
		int u = q[l++], e = hd[u];
		while (e != -1 && !dg[to[e]]) e = nx[e];
		int v = to[e];
		f0[v] += max(f0[u], f1[u]);
		f1[v] += f0[u];
		dg[u] = 0;
		dg[v]--;
		if (dg[v] == 1) q[r++] = v;
	}
	for (int i = 1; i <= n; i++) {
		if (dg[i]) cy[i] = 1;
	}
	long long ans = 0;
	for (int s = 1; s <= n; s++) if (cy[s] && !vs[s]) {
			int k = 0, u = s, pe = -1;
			while (1) {
				vs[u] = 1;
				cl[k++] = u;
				int e = hd[u];
				while (e != -1 && (!cy[to[e]] || (e >> 1) == pe)) e = nx[e];
				pe = e >> 1;
				u = to[e];
				if (u == s) break;
			}
			long long x = f0[cl[0]], y = -INF, z;
			for (int i = 1; i < k; i++) {
				long long a = max(x, y) + f0[cl[i]], b = x + f1[cl[i]];
				x = a;
				y = b;
			}
			z = max(x, y);
			x = -INF;
			y = f1[cl[0]];
			for (int i = 1; i < k; i++) {
				long long a = max(x, y) + f0[cl[i]], b = x + f1[cl[i]];
				x = a;
				y = b;
			}
			ans += max(z, x);
		}
	cout << ans << '\n';
	return 0;
}
```

## C. 桥梁不便指数

先找出基环树中的唯一环，因为只有环上的边可以删除。对每个环点做树形 DP，求出其挂树的最大深度，并记录所有挂树内部的最大直径。把环复制一遍，删除某条环边后就对应一个长度固定的链区间；链上跨挂树的最长路可写成 $((h_i-x_i)+(h_j+x_j))$。用线段树维护区间最大值并枚举删除哪条环边，取形成树的最小直径即可。

代码：

```cpp
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 200010, M = 400010;
const ll inf = -(1LL << 60);
struct node {
	ll a, b, c;
};
int n, m, ec, sz, hd[N], to[M], nx[M], dg[N], pa[N], od[N], sk[N], cv[N];
ll we[M], dp[N], ew[N], hh[N], xx[N * 2];
bool cy[N];
node tr[1 << 21];
void ad(int u, int v, ll w) {
	to[++ec] = v, we[ec] = w, nx[ec] = hd[u], hd[u] = ec, dg[u]++;
}
node mg(node x, node y) {
	return {max(x.a, y.a), max(x.b, y.b), max({x.c, y.c, x.a + y.b})};
}
node qu(int l, int r) {
	node x = {inf, inf, inf}, y = x;
	for (l += sz - 1, r += sz - 1; l <= r; l >>= 1, r >>= 1) {
		if (l & 1) x = mg(x, tr[l++]);
		if (!(r & 1)) y = mg(tr[r--], y);
	}
	return mg(x, y);
}
int main() {
	cin >> n;
	for (int i = 1, u, v; i <= n; i++) {
		ll w;
		cin >> u >> v >> w;
		ad(u, v, w), ad(v, u, w);
	}
	queue<int> q;
	for (int i = 1; i <= n; i++) cy[i] = 1;
	for (int i = 1; i <= n; i++) {
		if (dg[i] == 1) q.push(i);
	}
	while (q.size()) {
		int u = q.front();
		q.pop(), cy[u] = 0;
		for (int e = hd[u]; e; e = nx[e]) {
			int v = to[e];
			if (cy[v] && --dg[v] == 1) q.push(v);
		}
	}
	int rt = 0;
	for (int i = 1; i <= n && !rt; i++) {
		if (cy[i]) rt = i;
	}
	for (int u = rt, pr = 0;;) {
		cv[++m] = u;
		int v = 0;
		ll w = 0;
		for (int e = hd[u]; e; e = nx[e]) if (cy[to[e]] && to[e] != pr) {
			v = to[e], w = we[e];
			break;
		}
		ew[m] = w;
		if (v == rt)break;
		pr = u, u = v;
	}
	int tp = 0, on = 0;
	for (int i = 1; i <= n; i++) {
		if (cy[i]) sk[++tp] = i;
	}
	while (tp) {
		int u = sk[tp--];
		od[++on] = u;
		for (int e = hd[u]; e; e = nx[e]) {
			int v = to[e];
			if (!cy[v] && v != pa[u]) pa[v] = u, sk[++tp] = v;
		}
	}
	ll ds = 0;
	for (int i = on; i; i--) {
		int u = od[i];
		ll a = 0, b = 0;
		for (int e = hd[u]; e; e = nx[e]) {
			int v = to[e];
			if (pa[v] == u) {
				ll z = dp[v] + we[e];
				if (z > a)b = a, a = z;
				else if (z > b)b = z;
			}
		}
		dp[u] = a, ds = max(ds, a + b);
	}
	for (int i = 1; i <= m; i++)hh[i] = dp[cv[i]];
	xx[1] = 0;
	for (int i = 2; i <= m * 2; i++) xx[i] = xx[i - 1] + ew[(i - 2) % m + 1];
	for (sz = 1; sz < m * 2; sz <<= 1);
	for (int i = 1; i < sz * 2; i++) tr[i] = {inf, inf, inf};
	for (int i = 1; i <= m * 2; i++) {
		int j = (i - 1) % m + 1;
		tr[sz + i - 1] = {hh[j] - xx[i], hh[j] + xx[i], inf};
	}
	for (int i = sz - 1; i; i--) tr[i] = mg(tr[i << 1], tr[i << 1 | 1]);
	ll ans = 1LL << 62;
	for (int i = 1; i <= m; i++) {
		node z = qu(i + 1, i + m);
		ans = min(ans, max(ds, z.c));
	}
	cout << ans;
	return 0;
}
```
