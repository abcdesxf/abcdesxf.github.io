# 课程学习笔记 - 启发式合并

## A. 彩球

每个箱子只需维护其中出现过的颜色。每次合并时，将颜色数较少的集合插入颜色数较多的集合，再清空原箱子。由于集合大小至少翻倍，同一种颜色最多被移动 $O(\log N)$ 次。

代码：

```cpp
#include<bits/stdc++.h>
using namespace std;
const int M = 200010;
int a, b, n, q, c[M];
set<int> st[M];
int main() {
	ios::sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	cin >> n >> q;
	for (int i = 1; i <= n; i++) cin >> c[i], st[i].insert(c[i]);
	while (q--) {
		cin >> a >> b;
		if (st[a].size() > st[b].size()) swap(st[a], st[b]);
		for (int x : st[a]) st[b].insert(x);
		st[a].clear();
		cout << st[b].size() << '\n';
	}
	return 0;
}
```

## B. 梦幻布丁

维护每种颜色包含的位置集合和每个位置当前所属的内部颜色。初始答案是相邻颜色不同的位置对数量加一。合并两种颜色时只会消除它们之间的分界，因此遍历较小集合，检查其左右邻点是否属于目标集合并减少答案；若源集合更大，就交换两个颜色对应的内部编号后再合并。

代码：

```cpp
#include<bits/stdc++.h>
using namespace std;
const int M = 1000010;
int m, n, x, y, op, ans, a[100010], id[M];
set<int> st[M];
int main() {
	ios::sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	cin >> n >> m;
	for (int i = 1; i < M; i++) id[i] = i;
	for (int i = 1; i <= n; i++) cin >> a[i], st[a[i]].insert(i), ans += a[i] != a[i - 1];
	while (m--) {
		cin >> op;
		if (op == 2) {
			cout << ans << '\n';
			continue;
		}
		cin >> x >> y;
		if (x == y) continue;
		if (st[id[x]].size() > st[id[y]].size()) swap(id[x], id[y]);
		int u = id[x], v = id[y];
		for (int p : st[u]) ans -= a[p - 1] == v, ans -= a[p + 1] == v;
		for (int p : st[u]) a[p] = v, st[v].insert(p);
		st[u].clear();
	}
	return 0;
}
```

## C. Lomsat gelral

先求出每个节点的子树大小、重儿子和 DFS 序。处理节点时先计算并清空所有轻儿子的答案，再计算重儿子并保留其颜色统计，最后把所有轻儿子子树重新加入。每经过一条轻边，所在子树大小至少翻倍，因此每个节点只会被重复加入 $O(\log N)$ 次。

代码：

```cpp
#include<bits/stdc++.h>
using namespace std;
const int M = 100010;
int n, ec, mx, tt, c[M], ct[M], hd[M], hs[M], nx[M * 2], ot[M], sz[M], ti[M], to[M * 2], tr[M];
long long sm, an[M];
void add(int u, int v) {
	to[++ec] = v;
	nx[ec] = hd[u];
	hd[u] = ec;
}
void dfs(int u, int f) {
	ti[u] = ++tt;
	tr[tt] = u;
	sz[u] = 1;
	for (int e = hd[u]; e; e = nx[e]) {
		int v = to[e];
		if (v == f) continue;
		dfs(v, u);
		sz[u] += sz[v];
		if (sz[v] > sz[hs[u]]) hs[u] = v;
	}
	ot[u] = tt;
}
void upd(int u, int v) {
	int x = c[u];
	ct[x] += v;
	if (v < 0) return;
	if (ct[x] > mx) mx = ct[x], sm = x;
	else if (ct[x] == mx) sm += x;
}
void dsu(int u, int f, bool kp) {
	for (int e = hd[u]; e; e = nx[e]) {
		int v = to[e];
		if (v != f && v != hs[u]) dsu(v, u, 0);
	}
	if (hs[u]) dsu(hs[u], u, 1);
	for (int e = hd[u]; e; e = nx[e]) {
		int v = to[e];
		if (v == f || v == hs[u]) continue;
		for (int i = ti[v]; i <= ot[v]; i++) upd(tr[i], 1);
	}
	upd(u, 1);
	an[u] = sm;
	if (!kp) {
		for (int i = ti[u]; i <= ot[u]; i++) upd(tr[i], -1);
		mx = sm = 0;
	}
}
int main() {
	ios::sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	cin >> n;
	for (int i = 1; i <= n; i++) cin >> c[i];
	for (int i = 1, u, v; i < n; i++) cin >> u >> v, add(u, v), add(v, u);
	dfs(1, 0);
	dsu(1, 0, 1);
	for (int i = 1; i <= n; i++) cout << an[i] << ' ';
	return 0;
}
```

## D. 明辨是非

使用并查集维护已经确定相等的变量，每个连通块再用集合记录与它确定不等的连通块。加入不等关系时，只要两个变量不在同一连通块中就可以接受；加入相等关系时，若两个连通块之间已有不等关系则拒绝，否则把较小连通块合并到较大连通块，并同步修改所有相关的不等关系。

代码：

```cpp
#include<bits/stdc++.h>
using namespace std;
const int M = 200010;
int k, n, a, b, ra, rb, fa[M], op[100010], sz[M], x[100010], y[100010], z[M];
set<int> st[M];
int fd(int x) {
	if (fa[x] == x) return x;
	return fa[x] = fd(fa[x]);
}
int main() {
	ios::sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	cin >> n;
	for (int i = 1; i <= n; i++) cin >> x[i] >> y[i] >> op[i], z[i * 2 - 1] = x[i], z[i * 2] = y[i];
	sort(z + 1, z + n * 2 + 1);
	k = unique(z + 1, z + n * 2 + 1) - z - 1;
	for (int i = 1; i <= k; i++) fa[i] = i, sz[i] = 1;
	for (int i = 1; i <= n; i++) {
		a = lower_bound(z + 1, z + k + 1, x[i]) - z;
		b = lower_bound(z + 1, z + k + 1, y[i]) - z;
		ra = fd(a), rb = fd(b);
		if (!op[i]) {
			if (ra == rb) cout << "NO\n";
			else cout << "YES\n", st[ra].insert(rb), st[rb].insert(ra);
			continue;
		}
		if (ra == rb) {
			cout << "YES\n";
			continue;
		}
		if (st[ra].count(rb)) {
			cout << "NO\n";
			continue;
		}
		if (sz[ra] > sz[rb]) swap(ra, rb);
		fa[ra] = rb;
		sz[rb] += sz[ra];
		for (int v : st[ra]) {
			v = fd(v);
			st[v].erase(ra);
			st[v].insert(rb);
			st[rb].insert(v);
		}
		st[ra].clear();
		cout << "YES\n";
	}
	return 0;
}
```
