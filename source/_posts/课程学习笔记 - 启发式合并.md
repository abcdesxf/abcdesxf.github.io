# 课程学习笔记 - 启发式合并

## A. 彩球

每个箱子只需维护其中出现过的颜色。每次合并时，将颜色数较少的集合插入颜色数较多的集合，再清空原箱子。由于集合大小至少翻倍，同一种颜色最多被移动 $O(\log N)$ 次。

代码：

```cpp
#include<bits/stdc++.h>
using namespace std;
const int M = 200010;
int a, b, n, q, c[M];
set<int> st[M];
int main() {
	ios::sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	cin >> n >> q;
	for (int i = 1; i <= n; i++) cin >> c[i], st[i].insert(c[i]);
	while (q--) {
		cin >> a >> b;
		if (st[a].size() > st[b].size()) swap(st[a], st[b]);
		for (int x : st[a]) st[b].insert(x);
		st[a].clear();
		cout << st[b].size() << '\n';
	}
	return 0;
}
```

## B. 梦幻布丁

维护每种颜色包含的位置集合和每个位置当前所属的内部颜色。初始答案是相邻颜色不同的位置对数量加一。合并两种颜色时只会消除它们之间的分界，因此遍历较小集合，检查其左右邻点是否属于目标集合并减少答案；若源集合更大，就交换两个颜色对应的内部编号后再合并。

代码：

```cpp
#include<bits/stdc++.h>
using namespace std;
const int M = 1000010;
int m, n, x, y, op, ans, a[100010], id[M];
set<int> st[M];
int main() {
	ios::sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	cin >> n >> m;
	for (int i = 1; i < M; i++) id[i] = i;
	for (int i = 1; i <= n; i++) cin >> a[i], st[a[i]].insert(i), ans += a[i] != a[i - 1];
	while (m--) {
		cin >> op;
		if (op == 2) {
			cout << ans << '\n';
			continue;
		}
		cin >> x >> y;
		if (x == y) continue;
		if (st[id[x]].size() > st[id[y]].size()) swap(id[x], id[y]);
		int u = id[x], v = id[y];
		for (int p : st[u]) ans -= a[p - 1] == v, ans -= a[p + 1] == v;
		for (int p : st[u]) a[p] = v, st[v].insert(p);
		st[u].clear();
	}
	return 0;
}
```

## C. Lomsat gelral

先求出每个节点的子树大小、重儿子和 DFS 序。处理节点时先计算并清空所有轻儿子的答案，再计算重儿子并保留其颜色统计，最后把所有轻儿子子树重新加入。每经过一条轻边，所在子树大小至少翻倍，因此每个节点只会被重复加入 $O(\log N)$ 次。

代码：

```cpp
#include<bits/stdc++.h>
using namespace std;
const int M = 100010;
int n, ec, mx, tt, c[M], ct[M], hd[M], hs[M], nx[M * 2], ot[M], sz[M], ti[M], to[M * 2], tr[M];
long long sm, an[M];
void add(int u, int v) {
	to[++ec] = v;
	nx[ec] = hd[u];
	hd[u] = ec;
}
void dfs(int u, int f) {
	ti[u] = ++tt;
	tr[tt] = u;
	sz[u] = 1;
	for (int e = hd[u]; e; e = nx[e]) {
		int v = to[e];
		if (v == f) continue;
		dfs(v, u);
		sz[u] += sz[v];
		if (sz[v] > sz[hs[u]]) hs[u] = v;
	}
	ot[u] = tt;
}
void upd(int u, int v) {
	int x = c[u];
	ct[x] += v;
	if (v < 0) return;
	if (ct[x] > mx) mx = ct[x], sm = x;
	else if (ct[x] == mx) sm += x;
}
void dsu(int u, int f, bool kp) {
	for (int e = hd[u]; e; e = nx[e]) {
		int v = to[e];
		if (v != f && v != hs[u]) dsu(v, u, 0);
	}
	if (hs[u]) dsu(hs[u], u, 1);
	for (int e = hd[u]; e; e = nx[e]) {
		int v = to[e];
		if (v == f || v == hs[u]) continue;
		for (int i = ti[v]; i <= ot[v]; i++) upd(tr[i], 1);
	}
	upd(u, 1);
	an[u] = sm;
	if (!kp) {
		for (int i = ti[u]; i <= ot[u]; i++) upd(tr[i], -1);
		mx = sm = 0;
	}
}
int main() {
	ios::sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	cin >> n;
	for (int i = 1; i <= n; i++) cin >> c[i];
	for (int i = 1, u, v; i < n; i++) cin >> u >> v, add(u, v), add(v, u);
	dfs(1, 0);
	dsu(1, 0, 1);
	for (int i = 1; i <= n; i++) cout << an[i] << ' ';
	return 0;
}
```

## D. 明辨是非

使用并查集维护已经确定相等的变量，每个连通块再用集合记录与它确定不等的连通块。加入不等关系时，只要两个变量不在同一连通块中就可以接受；加入相等关系时，若两个连通块之间已有不等关系则拒绝，否则把较小连通块合并到较大连通块，并同步修改所有相关的不等关系。

代码：

```cpp
#include<bits/stdc++.h>
using namespace std;
const int M = 200010;
int k, n, a, b, ra, rb, fa[M], op[100010], sz[M], x[100010], y[100010], z[M];
set<int> st[M];
int fd(int x) {
	if (fa[x] == x) return x;
	return fa[x] = fd(fa[x]);
}
int main() {
	ios::sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	cin >> n;
	for (int i = 1; i <= n; i++) cin >> x[i] >> y[i] >> op[i], z[i * 2 - 1] = x[i], z[i * 2] = y[i];
	sort(z + 1, z + n * 2 + 1);
	k = unique(z + 1, z + n * 2 + 1) - z - 1;
	for (int i = 1; i <= k; i++) fa[i] = i, sz[i] = 1;
	for (int i = 1; i <= n; i++) {
		a = lower_bound(z + 1, z + k + 1, x[i]) - z;
		b = lower_bound(z + 1, z + k + 1, y[i]) - z;
		ra = fd(a), rb = fd(b);
		if (!op[i]) {
			if (ra == rb) cout << "NO\n";
			else cout << "YES\n", st[ra].insert(rb), st[rb].insert(ra);
			continue;
		}
		if (ra == rb) {
			cout << "YES\n";
			continue;
		}
		if (st[ra].count(rb)) {
			cout << "NO\n";
			continue;
		}
		if (sz[ra] > sz[rb]) swap(ra, rb);
		fa[ra] = rb;
		sz[rb] += sz[ra];
		for (int v : st[ra]) {
			v = fd(v);
			st[v].erase(ra);
			st[v].insert(rb);
			st[rb].insert(v);
		}
		st[ra].clear();
		cout << "YES\n";
	}
	return 0;
}
```

## E. 异或树

令根到节点 $u$ 的点权异或和为 $xr_u$。若两点 $x,y$ 的最近公共祖先是 $u$，则路径异或和为 $xr_x\oplus xr_y\oplus a_u$，所以零异或路径等价于 $xr_x\oplus xr_y=a_u$。自底向上维护每棵子树中尚未被修改节点隔断的异或和集合，合并儿子时检查两个集合中是否存在异或值相差 $a_u$ 的元素。若存在，就必须在这些路径上修改节点，贪心修改它们共同经过且深度最小的 $u$，随后清空当前集合；否则使用启发式合并保留所有异或和。每个元素只会被移动 $O(\log N)$ 次。

代码：

```cpp
#include<bits/stdc++.h>
using namespace std;
const int M = 200010;
int n, ec, tp, tt, ans, a[M], fa[M], hd[M], nx[M * 2], od[M], sk[M], to[M * 2], xr[M];
set<int> st[M];
void add(int u, int v) {
	to[++ec] = v;
	nx[ec] = hd[u];
	hd[u] = ec;
}
int main() {
	ios::sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	cin >> n;
	for (int i = 1; i <= n; i++) cin >> a[i];
	for (int i = 1, u, v; i < n; i++) cin >> u >> v, add(u, v), add(v, u);
	sk[++tp] = 1, xr[1] = a[1];
	while (tp) {
		int u = sk[tp--];
		od[++tt] = u;
		for (int e = hd[u]; e; e = nx[e]) {
			int v = to[e];
			if (v == fa[u]) continue;
			fa[v] = u, xr[v] = xr[u] ^ a[v], sk[++tp] = v;
		}
	}
	for (int i = n; i; i--) {
		int u = od[i];
		bool ok = 1;
		st[u].insert(xr[u]);
		for (int e = hd[u]; e; e = nx[e]) {
			int v = to[e];
			if (fa[v] != u) continue;
			if (st[u].size() < st[v].size()) swap(st[u], st[v]);
			for (int x : st[v]) if (st[u].count(x ^ a[u])) {
				ok = 0;
				break;
			}
			if (!ok) break;
			for (int x : st[v]) st[u].insert(x);
			st[v].clear();
		}
		if (!ok) ans++, st[u].clear();
	}
	cout << ans;
	return 0;
}
```

## F. 树上请求

一棵子树在 DFS 序中对应连续区间。按照深度分别保存节点的 DFS 序，并维护该深度中字母状态的前缀异或和，其中第 $k$ 位表示字母是否出现奇数次。查询时在指定深度的有序 DFS 序中二分出子树区间，再由前缀异或得到所有字母的奇偶状态。至多一个字母出现奇数次时可以重排成回文串。

代码：

```cpp
#include<bits/stdc++.h>
using namespace std;
const int M = 500010;
int n, m, ec, tp, tt, bg[M], ct[M], cu[M], dp[M], fa[M], hd[M], nx[M], od[M], ps[M], sk[M], sz[M], ti[M], to[M], xr[M];
char s[M];
void add(int u, int v) {
	to[++ec] = v;
	nx[ec] = hd[u];
	hd[u] = ec;
}
int main() {
	ios::sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	cin >> n >> m;
	dp[1] = 1;
	for (int i = 2; i <= n; i++) cin >> fa[i], dp[i] = dp[fa[i]] + 1, add(fa[i], i);
	cin >> s;
	sk[++tp] = 1;
	while (tp) {
		int u = sk[tp--];
		od[++tt] = u, ti[u] = tt, ct[dp[u]]++;
		for (int e = hd[u]; e; e = nx[e]) sk[++tp] = to[e];
	}
	for (int i = 1; i <= n; i++) sz[i] = 1;
	for (int i = n; i > 1; i--) sz[fa[od[i]]] += sz[od[i]];
	bg[1] = 1;
	for (int i = 2; i <= n; i++) bg[i] = bg[i - 1] + ct[i - 1];
	for (int i = 1; i <= n; i++) cu[i] = bg[i];
	for (int i = 1; i <= n; i++) {
		int u = od[i], d = dp[u], p = cu[d]++;
		ps[p] = i, xr[p] = (p == bg[d] ? 0 : xr[p - 1]) ^ (1 << (s[u - 1] - 'a'));
	}
	while (m--) {
		int v, h;
		cin >> v >> h;
		int l = lower_bound(ps + bg[h], ps + bg[h] + ct[h], ti[v]) - ps;
		int r = upper_bound(ps + bg[h], ps + bg[h] + ct[h], ti[v] + sz[v] - 1) - ps - 1;
		int x = l > r ? 0 : xr[r] ^ (l == bg[h] ? 0 : xr[l - 1]);
		cout << (__builtin_popcount(x) <= 1 ? "Yes\n" : "No\n");
	}
	return 0;
}
```

## G. Vicky 的送货服务

把彩虹路的边从起点开始每两条分成一组，则每组都由两条同色边组成，末尾还可以额外保留一条边。对于同一城市、同一颜色的所有邻点，任意两个邻点之间都能走完一组，因此用并查集把它们合并，表示可以通过若干完整的两边组互相到达。每个并查集再维护所有成员通过一条道路可以直接到达的城市集合，查询 $x$ 到 $y$ 时，只需判断二者是否属于同一并查集，或 $y$ 是否在 $x$ 所在集合的一步邻点中。加入道路时按“城市和颜色”找到已有邻点并合并，集合使用启发式合并。

代码：

```cpp
#include<bits/stdc++.h>
using namespace std;
const int M = 100010;
int n, m, c, q, fa[M];
set<int> st[M];
unordered_map<long long, int> mp;
int fd(int x) {
	if (fa[x] == x) return x;
	return fa[x] = fd(fa[x]);
}
void un(int x, int y) {
	x = fd(x), y = fd(y);
	if (x == y) return;
	if (st[x].size() > st[y].size()) swap(x, y);
	fa[x] = y;
	for (int v : st[x]) st[y].insert(v);
	st[x].clear();
}
long long ky(int x, int z) {
	return 1LL * x * (c + 1) + z;
}
void add(int x, int y, int z) {
	int u = x, v = y;
	long long a = ky(x, z), b = ky(y, z);
	auto it = mp.find(a);
	if (it == mp.end()) mp[a] = y;
	else un(y, it->second);
	it = mp.find(b);
	if (it == mp.end()) mp[b] = x;
	else un(x, it->second);
	st[fd(u)].insert(v), st[fd(v)].insert(u);
}
int main() {
	ios::sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	cin >> n >> m >> c >> q;
	mp.reserve((m + q) * 4);
	mp.max_load_factor(0.7);
	for (int i = 1; i <= n; i++) fa[i] = i;
	for (int i = 1, x, y, z; i <= m; i++) cin >> x >> y >> z, add(x, y, z);
	while (q--) {
		char op;
		int x, y, z;
		cin >> op >> x >> y;
		if (op == '+') cin >> z, add(x, y, z);
		else cout << (fd(x) == fd(y) || st[fd(x)].count(y) ? "Yes\n" : "No\n");
	}
	return 0;
}
```
