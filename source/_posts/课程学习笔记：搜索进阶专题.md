# 课程学习笔记：搜索进阶专题

## A. 八数码问题

经典的八数码。

从目标状态 `12345678x` 出发做一次反向 BFS，预处理所有能够到达目标的状态。搜索时用康托展开把每个排列映射成唯一编号，并记录该状态走向目标时的下一步操作。对于每组输入，若没有被 BFS 访问到就输出 `unsolvable`，否则沿着预处理的操作序列输出答案。

代码：

```cpp
#include <bits/stdc++.h>
using namespace std;
const int N = 362881;
int fct[10], pre[N], qid[N], hh, tt;
unsigned char q[N][9], zp[N], a[9], b[9];
char op[N], ch;
int dx[4] = {-1, 1, 0, 0}, dy[4] = {0, 0, -1, 1};
char rev[4] = {'d', 'u', 'r', 'l'};
int ct(unsigned char s[]) {
    int res = 0;
    for (int i = 0; i < 9; i++) {
        int cnt = 0;
        for (int j = i + 1; j < 9; j++) {
            if (s[j] < s[i]) {
                cnt++;
            }
        }
        res += cnt * fct[8 - i];
    }
    return res;
}
void bfs() {
    memset(pre, -1, sizeof(pre));
    for (int i = 0; i < 8; i++) {
        a[i] = i + 1;
    }
    a[8] = 0;
    int id = ct(a);
    pre[id] = id;
    memcpy(q[tt], a, 9);
    qid[tt] = id;
    zp[tt] = 8;
    tt++;
    while (hh < tt) {
        int u = qid[hh], x = zp[hh], r = x / 3, c = x % 3;
        for (int k = 0; k < 4; k++) {
            int nr = r + dx[k], nc = c + dy[k];
            if (nr < 0 || nr >= 3 || nc < 0 || nc >= 3) {
                continue;
            }
            int nx = nr * 3 + nc;
            memcpy(b, q[hh], 9);
            swap(b[x], b[nx]);
            int v = ct(b);
            if (pre[v] != -1) {
                continue;
            }
            pre[v] = u;
            op[v] = rev[k];
            memcpy(q[tt], b, 9);
            qid[tt] = v;
            zp[tt] = nx;
            tt++;
        }
        hh++;
    }
}
int main() {
    fct[0] = 1;
    for (int i = 1; i <= 9; i++) {
        fct[i] = fct[i - 1] * i;
    }
    bfs();
    while (cin >> ch) {
        a[0] = (ch == 'x' ? 0 : ch - '0');
        for (int i = 1; i < 9; i++) {
            cin >> ch;
            a[i] = (ch == 'x' ? 0 : ch - '0');
        }
        int id = ct(a);
        if (pre[id] == -1) {
            cout << "unsolvable\n";
            continue;
        }
        while (pre[id] != id) {
            cout << op[id];
            id = pre[id];
        }
        cout << '\n';
    }
    return 0;
}
```

## B. 推箱子

把“箱子位置 + 人的位置”作为一个状态，每次人向四个方向移动：若前方不是箱子，代价为 `(推箱次数不变，总步数+1)`；若推动箱子，代价为 `(推箱次数+1，总步数+1)`。使用 Dijkstra，并按“推箱次数优先、总操作次数次优”比较距离，第一个箱子到达目标的状态就是答案；记录前驱和操作字符即可还原路径。状态数最多为 $(rc)^2=160000$。

代码：

```cpp
#include <bits/stdc++.h>
using namespace std;
const int MAXN = 410, MAXV = 160010, MAXH = 800010, INF = 0x3f3f3f3f;
struct Node {
    int id, p, s;
};
int r, c, n, sp, bp, tp, hs, dp[MAXV], ds[MAXV], pre[MAXV];
int dx[4] = {-1, 1, 0, 0}, dy[4] = {0, 0, 1, -1};
char g[25][25], way[MAXV], ans[MAXV];
char lo[4] = {'n', 's', 'e', 'w'}, up[4] = {'N', 'S', 'E', 'W'};
Node hp[MAXH];
bool cmp(Node a, Node b) {
    if (a.p != b.p) return a.p < b.p;
    return a.s < b.s;
}
void push(Node x) {
    int i = ++hs;
    while (i > 1 && cmp(x, hp[i / 2])) {
        hp[i] = hp[i / 2];
        i /= 2;
    }
    hp[i] = x;
}
Node pop() {
    Node res = hp[1];
    Node x = hp[hs--];
    int i = 1;
    while (i * 2 <= hs) {
        int j = i * 2;
        if (j + 1 <= hs && cmp(hp[j + 1], hp[j])) {
            j++;
        }
        if (!cmp(hp[j], x)) break;
        hp[i] = hp[j];
        i = j;
    }
    if (hs >= 1) hp[i] = x;
    return res;
}
bool check(int x, int y) { return x >= 0 && x < r && y >= 0 && y < c && g[x][y] != '#'; }
bool more(int p, int s, int id) {
    if (p != dp[id]) return p < dp[id];
    return s < ds[id];
}
void print(int ed) {
    int len = 0;
    while (pre[ed] != -1) {
        ans[len++] = way[ed];
        ed = pre[ed];
    }
    for (int i = len - 1; i >= 0; i--) {
        cout << ans[i];
    }
    cout << '\n';
}
void solve() {
    n = r * c;
    hs = 0;
    memset(dp, 0x3f, sizeof(dp));
    memset(ds, 0x3f, sizeof(ds));
    memset(pre, -1, sizeof(pre));
    int st = bp * n + sp;
    dp[st] = 0;
    ds[st] = 0;
    push({st, 0, 0});
    while (hs) {
        Node now = pop();
        int id = now.id;
        if (now.p != dp[id] || now.s != ds[id]) {
            continue;
        }
        int b = id / n, m = id % n;
        if (b == tp) {
            print(id);
            return;
        }
        int bx = b / c, by = b % c, mx = m / c, my = m % c;
        for (int d = 0; d < 4; d++) {
            int nx = mx + dx[d], ny = my + dy[d];
            if (!check(nx, ny)) continue;
            int nm = nx * c + ny, nb = b, np = now.p;
            char ch = lo[d];
            if (nm == b) {
                int tx = bx + dx[d], ty = by + dy[d];
                if (!check(tx, ty)) continue;
                nb = tx * c + ty;
                np++;
                ch = up[d];
            }
            int nid = nb * n + nm, ns = now.s + 1;
            if (more(np, ns, nid)) {
                dp[nid] = np, ds[nid] = ns, pre[nid] = id, way[nid] = ch;
                push({nid, np, ns});
            }
        }
    }
    cout << "Impossible.\n";
}
int main() {
    while (cin >> r >> c) {
        if (r == 0 && c == 0) break;
        for (int i = 0; i < r; i++) {
            cin >> g[i];
            for (int j = 0; j < c; j++) {
                int pos = i * c + j;
                if (g[i][j] == 'S') sp = pos;
                if (g[i][j] == 'B') bp = pos;
                if (g[i][j] == 'T') tp = pos;
            }
        }
        solve();
    }
    return 0;
}
```

## C. 第二十一课：恶梦

王子和公主同时搜索：每一秒先把时间加一，再让王子扩展 3 层、公主扩展 1 层。恶魔每秒可以扩散曼哈顿距离 2，因此在第 `t` 秒，一个格子到任意恶魔的曼哈顿距离若不超过 `2t`，该格子就不能进入，墙也挡不住恶魔扩散。两边分别记录访问状态，只要一边扩展到另一边访问过的安全格子，就说明能在当前秒相遇；两个队列都为空时仍未相遇则无解。

代码：

```cpp
#include <bits/stdc++.h>
using namespace std;
const int N = 805, V = N * N;
int n, m, zx[2], zy[2], q[2][V], hh[2], tt[2], v[2][N][N];
int dx[4] = {-1, 1, 0, 0}, dy[4] = {0, 0, -1, 1};
char g[N][N];
bool safe(int x, int y, int t) {
    for (int i = 0; i < 2; i++) {
        if (abs(x - zx[i]) + abs(y - zy[i]) <= 2 * t) {
            return false;
        }
    }
    return true;
}
bool bfs(int p, int t) {
    int len = tt[p] - hh[p];
    while (len--) {
        int u = q[p][hh[p]++], x = u / m, y = u % m;
        if (!safe(x, y, t)) {
            continue;
        }
        for (int d = 0; d < 4; d++) {
            int nx = x + dx[d], ny = y + dy[d];
            if (nx < 0 || nx >= n || ny < 0 || ny >= m || g[nx][ny] == 'X') {
                continue;
            }
            if (!safe(nx, ny, t) || v[p][nx][ny]) {
                continue;
            }
            if (v[p ^ 1][nx][ny]) {
                return true;
            }
            v[p][nx][ny] = 1;
            q[p][tt[p]++] = nx * m + ny;
        }
    }
    return false;
}
int solve() {
    memset(v, 0, sizeof(v));
    hh[0] = hh[1] = tt[0] = tt[1] = 0;
    int z = 0;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (g[i][j] == 'M') {
                v[0][i][j] = 1;
                q[0][tt[0]++] = i * m + j;
            } else if (g[i][j] == 'G') {
                v[1][i][j] = 1;
                q[1][tt[1]++] = i * m + j;
            } else if (g[i][j] == 'Z') {
                zx[z] = i, zy[z++] = j;
            }
        }
    }
    for (int t = 1; hh[0] < tt[0] || hh[1] < tt[1]; t++) {
        for (int i = 0; i < 3; i++) {
            if (bfs(0, t)) {
                return t;
            }
        }
        if (bfs(1, t)) {
            return t;
        }
    }
    return -1;
}
int main() {
    int T;
    cin >> T;
    while (T--) {
        cin >> n >> m;
        for (int i = 0; i < n; i++) {
            cin >> g[i];
        }
        cout << solve() << '\n';
    }
    return 0;
}
```

## D. 第二十一课：k短路

先在反图上从终点跑 Dijkstra，得到每个点到终点的最短距离 `dis`，它就是 A* 的估价函数。再从起点进行 A*，状态记录当前真实路程 `d`，并按 `d+dis[u]` 从小到大取出；一个点第几次出队，就表示找到了到该点的第几短走法，所以终点第 `k` 次出队时的 `d` 就是答案。每个点最多扩展 `k` 次，不可到达终点的点不加入队列；若 `s=t`，需要跳过长度为 0 的空路径，因此把 `k` 加一。

代码：

```cpp
#include <bits/stdc++.h>
using namespace std;
const int N = 1010, M = 100010, INF = 0x3f3f3f3f;
char gc() {
    static const int S = 1 << 20;
    static char buf[S];
    static int p = 0, len = 0;
    if (p == len) {
        len = fread(buf, 1, S, stdin);
        p = 0;
        if (len == 0) {
            return 0;
        }
    }
    return buf[p++];
}
int read() {
    int x = 0;
    char c = gc();
    while (c < '0' || c > '9') {
        c = gc();
    }
    while (c >= '0' && c <= '9') {
        x = x * 10 + c - '0';
        c = gc();
    }
    return x;
}
struct Node {
    int u, d, f;
    bool operator<(const Node &x) const {
        if (f != x.f) return f > x.f;
        return d < x.d;
    }
};
struct DNode {
    int u, d;
    bool operator<(const DNode &x) const {
        return d > x.d;
    }
};
int n, m, s, t, k, h1[N], h2[N], e1[M], e2[M], ne1[M], ne2[M], w1[M], w2[M], idx1, idx2, cnt[N], dis[N];
void add1(int a, int b, int c) {
    e1[idx1] = b;
    w1[idx1] = c;
    ne1[idx1] = h1[a];
    h1[a] = idx1++;
}
void add2(int a, int b, int c) {
    e2[idx2] = b;
    w2[idx2] = c;
    ne2[idx2] = h2[a];
    h2[a] = idx2++;
}
void dij() {
    priority_queue<DNode> q;
    memset(dis, 0x3f, sizeof dis);
    dis[t] = 0;
    q.push({t, 0});
    while (!q.empty()) {
        DNode now = q.top();
        q.pop();
        int u = now.u, d = now.d;
        if (d != dis[u]) {
            continue;
        }
        for (int i = h2[u]; i != -1; i = ne2[i]) {
            int v = e2[i];
            int nd = d + w2[i];
            if (dis[v] > nd) {
                dis[v] = nd;
                q.push({v, nd});
            }
        }
    }
}
int astar() {
    if (dis[s] == INF) {
        return -1;
    }
    int tgt = k + (s == t);
    priority_queue<Node> q;
    q.push({s, 0, dis[s]});
    while (!q.empty()) {
        Node now = q.top();
        q.pop();
        int u = now.u;
        if (cnt[u] >= tgt) {
            continue;
        }
        cnt[u]++;
        if (u == t && cnt[u] == tgt) {
            return now.d;
        }
        for (int i = h1[u]; i != -1; i = ne1[i]) {
            int v = e1[i];
            if (dis[v] == INF || cnt[v] >= tgt) {
                continue;
            }
            int nd = now.d + w1[i];
            q.push({v, nd, nd + dis[v]});
        }
    }
    return -1;
}
int main() {
    n = read(), m = read();
    memset(h1, -1, sizeof h1);
    memset(h2, -1, sizeof h2);
    for (int i = 0; i < m; i++) {
        int a = read(), b = read(), c = read();
        add1(a, b, c);
        add2(b, a, c);
    }
    s = read(), t = read(), k = read();
    dij();
    printf("%d\n", astar());
    return 0;
}
```

## E. 加加链

使用迭代加深搜索，依次枚举链的长度，第一次找到的答案就是最短加加链。搜索下一项时，枚举前面任意两项之和，并保证新数严格递增且不超过 $n$。若当前最大值即使在剩余步数中每次翻倍也无法达到 $n$，则直接剪枝。

代码：

```cpp
#include<bits/stdc++.h>
using namespace std;
int n, a[20];
bool dfs(int u, int d) {
	if (u == d) return a[u] == n;
	if ((a[u] << (d - u)) < n) return 0;
	bool v[205] = {};
	for (int i = u; i >= 0; i--) for (int j = i; j >= 0; j--) {
			int x = a[i] + a[j];
			if (x <= a[u] || x > n || v[x]) continue;
			v[x] = 1, a[u + 1] = x;
			if (dfs(u + 1, d)) return 1;
		}
	return 0;
}
int main() {
	ios::sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	while (cin >> n && n) {
		a[0] = 1;
		int d = 0;
		while (!dfs(0, d)) d++;
		for (int i = 0; i <= d; i++) cout << a[i] << (i == d ? '\n' : ' ');
	}
	return 0;
}
```
